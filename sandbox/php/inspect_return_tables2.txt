CREATE VIEW dbo.shopee_orders_cn_date_from_shipped_status_from_latest
AS
SELECT        a.OrderID, a.order_sn, a.taxinvoiceID, a.create_time, a.order_status, b.order_status AS latest_status, a.update_time AS invoice_date, b.update_time AS lasted_update, a.total_amount, a.is_return
FROM            dbo.shopee_orders AS a INNER JOIN
                         dbo.shopee_orders AS b ON a.order_sn = b.order_sn
WHERE        (a.order_status = 'PROCESSED') AND (b.order_status = 'CANCELLED') OR
                         (a.order_status = 'PROCESSED') AND (b.order_status = 'COMPLETED') AND (a.is_return = 1)

AS
